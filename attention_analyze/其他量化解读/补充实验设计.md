### 实验1：位置偏置 vs 内容相关

```python
# 对每个序列长度，采样10个不同内容的样本

seq_len = 74
num_samples = 10

results_by_head = {i: [] for i in range(8)}

for seed in range(num_samples):
    # 采样不同的物品序列
    item_seq = sample_item_seq(seq_len, seed=seed)
    attention_weights = get_attention_weights(item_seq)

    # 记录每个Head的热点位置
    for head_id in range(8):
        hotspots = find_hotspot_columns(attention_weights[head_id])
        results_by_head[head_id].append({
            'seed': seed,
            'top1_hotspot': hotspots[0].column,
            'top1_attention': hotspots[0].total_attention,
            'item_seq': item_seq,  # 保存用于后续分析
        })

# 分析：同一Head在不同样本下的热点位置变化
for head_id in range(8):
    top1_positions = [r['top1_hotspot'] for r in results_by_head[head_id]]

    # 计算位置的方差
    position_variance = np.var(top1_positions)

    if position_variance < 1.0:  # 几乎不变
        print(f"Head {head_id}: 位置偏置主导")
        print(f"  总是关注位置 {np.mean(top1_positions):.1f}")
    else:
        print(f"Head {head_id}: 内容相关")
        print(f"  热点位置均值为 {np.mean(top1_positions):.1f}, 方差为 {position_variance:.1f}")

        # 进一步分析：热点位置对应的物品特征
        analyze_hotspot_content(results_by_head[head_id])
```

### 分析热点位置的内容特征

```python
def analyze_hotspot_content(results):
    """分析热点位置对应的物品特征"""
    behavior_counts = {'pv': 0, 'cart': 0, 'fav': 0, 'buy': 0}
    category_counts = {}

    for r in results:
        hotspot_pos = r['top1_hotspot']
        hotspot_item = r['item_seq'][hotspot_pos]

        behavior_counts[hotspot_item['behavior_type']] += 1
        category_id = hotspot_item['category_id']
        category_counts[category_id] = category_counts.get(category_id, 0) + 1

    print("  热点位置的行为类型分布:")
    for behavior, count in behavior_counts.items():
        print(f"    {behavior}: {count}/{len(results)} ({count/len(results):.1%})")

    print("  热点位置的类别多样性:", len(category_counts))
```

**预期结果示例**：

```
Head 0: 位置偏置主导
  总是关注位置 73.0（最后位置）

Head 1: 内容相关
  热点位置均值为 35.2, 方差为 312.5
  热点位置的行为类型分布:
    pv: 3/10 (30%)
    cart: 1/10 (10%)
    fav: 2/10 (20%)
    buy: 4/10 (40%)  ← 购买行为占主导
  热点位置的类别多样性: 7
```

## 实验2：Query视角分析

### 分析每个Query位置的注意力分配

```python
def analyze_query_patterns_for_head(attn, head_id):
    """分析单个Head的Query模式"""
    seq_len = attn.shape[0]

    query_stats = []
    for query_pos in range(seq_len - 1):  # 排除目标位置
        row = attn[query_pos, :-1]  # Query对历史Key的关注

        entropy = compute_entropy(row)
        top_key = row.argmax().item()
        distance_to_target = (seq_len - 1) - top_key

        query_stats.append({
            'query_pos': query_pos,
            'entropy': entropy,
            'top_key': top_key,
            'distance_to_target': distance_to_target,
        })

    # 分析Query一致性
    top_keys = [s['top_key'] for s in query_stats]
    if len(set(top_keys)) == 1:
        print(f"Head {head_id}: 所有Query都关注位置 {top_keys[0]}")
        print(f"  → 目标导向型")
    else:
        # 检查是否关注相对位置
        top_keys_offset = [s['top_key'] - s['query_pos'] for s in query_stats]
        if np.std(top_keys_offset) < 2:
            print(f"Head {head_id}: 每个Query关注相对位置 {np.mean(top_keys_offset):.1f}")
            print(f"  → 局部平滑型")
        else:
            print(f"Head {head_id}: 不同Query关注不同位置")
            print(f"  → 查询特定型（可能内容相关）")

    return query_stats
```

**预期结果示例**：

```
Head 0: 所有Query都关注位置 73
  → 目标导向型
  熵值: 所有Query的熵都很低（~0.1）

Head 1: 不同Query关注不同位置
  → 查询特定型
  熵值: 部分Query熵高，部分熵低

Head 2: 每个Query关注相对位置 +1
  → 局部平滑型
  熵值: 中等（~1.5）
```

## 实验3：多头相似度分析

### 计算不同Head之间的注意力模式相似度

```python
def compute_head_similarity(attention_weights):
    """计算多头之间的相似度矩阵"""
    num_heads = attention_weights.shape[0]

    similarity_matrix = np.zeros((num_heads, num_heads))

    for i in range(num_heads):
        for j in range(num_heads):
            # 方法1：皮尔逊相关（扁平化后计算）
            flat_i = attention_weights[i].flatten().numpy()
            flat_j = attention_weights[j].flatten().numpy()

            corr = np.corrcoef(flat_i, flat_j)[0, 1]
            similarity_matrix[i, j] = corr

    return similarity_matrix

# 可视化
similarity = compute_head_similarity(attention_weights)
plt.imshow(similarity, cmap='RdYlGn', vmin=-1, vmax=1)
plt.colorbar()
plt.title('Head Similarity Matrix')
plt.xlabel('Head ID')
plt.ylabel('Head ID')

# 聚类
from sklearn.cluster import AgglomerativeClustering
clustering = AgglomerativeClustering(n_clusters=None, distance_threshold=0.5)
labels = clustering.fit_predict(1 - similarity)  # 用距离矩阵聚类

print("Head聚类结果:")
for cluster_id in range(max(labels) + 1):
    heads_in_cluster = np.where(labels == cluster_id)[0].tolist()
    print(f"  聚类{cluster_id}: {heads_in_cluster}")
```

**预期结果示例**：

```
Head聚类结果:
  聚类0: [0, 3, 4, 5, 6]  ← 稀疏型头（都关注最后位置）
  聚类1: [1, 7]           ← 分散型头（均匀分散）
  聚类2: [2]              ← 中间型
```

## 完整的分析流程

```python
def comprehensive_attention_analysis(seq_len=74, num_samples=10):
    """完整的注意力分析"""

    # ========== 第1步：多样本采样 ==========
    all_samples = []
    for seed in range(num_samples):
        item_seq = sample_item_seq(seq_len, seed=seed)
        attention_weights = get_attention_weights(item_seq)
        all_samples.append({
            'seed': seed,
            'item_seq': item_seq,
            'attention': attention_weights,
        })

    # ========== 第2步：位置偏置 vs 内容相关 ==========
    print("=" * 60)
    print("分析1: 位置偏置 vs 内容相关")
    print("=" * 60)

    for head_id in range(8):
        results = []
        for sample in all_samples:
            hotspots = find_hotspot_columns(sample['attention'][head_id])
            results.append({
                'seed': sample['seed'],
                'top1_pos': hotspots[0].column,
                'item_seq': sample['item_seq'],
            })

        top1_positions = [r['top1_pos'] for r in results]
        variance = np.var(top1_positions)

        if variance < 1.0:
            print(f"\nHead {head_id}: 位置偏置（总是关注位置 {np.mean(top1_positions):.0f}）")
        else:
            print(f"\nHead {head_id}: 内容相关（位置方差 = {variance:.1f}）")
            analyze_hotspot_content(results)

    # ========== 第3步：Query视角分析 ==========
    print("\n" + "=" * 60)
    print("分析2: Query视角（每个位置如何分配注意力）")
    print("=" * 60)

    # 用第一个样本做详细分析
    sample_attn = all_samples[0]['attention']
    for head_id in range(8):
        print(f"\nHead {head_id}:")
        analyze_query_patterns_for_head(sample_attn[head_id], head_id)

    # ========== 第4步：多头相似度 ==========
    print("\n" + "=" * 60)
    print("分析3: 多头相似度")
    print("=" * 60)

    similarity = compute_head_similarity(sample_attn)
    clustering = AgglomerativeClustering(n_clusters=None, distance_threshold=0.5)
    labels = clustering.fit_predict(1 - similarity)

    print("Head聚类:")
    for cluster_id in range(max(labels) + 1):
        heads = np.where(labels == cluster_id)[0].tolist()
        print(f"  组{cluster_id}: {heads}")

    return all_samples, similarity
```

## 关键指标总结

| 指标 | 计算方法 | 意义 |
|------|---------|------|
| **位置方差** | 同一Head在不同样本下的热点位置方差 | < 1.0 → 位置偏置<br>> 10 → 内容相关 |
| **Query一致性** | 所有Query的argmax Key是否相同 | 相同 → 目标导向<br>不同 → 查询特定 |
| **熵值** | 每行的熵 | < 1.0 → 稀疏<br>> log(seq_len)*0.8 → 分散 |
| **Head相似度** | 扁平化后的皮尔逊相关 | > 0.8 → 冗余<br>< 0.3 → 互补 |

## 结果解读指南

### 场景1：所有Head都是位置偏置

```
所有Head都总是关注最后位置
```

**解读**：
- 你的训练目标可能过于简单（只用历史预测目标）
- 模型没有学习到历史位置之间的复杂关系
- 建议：
  - 检查训练目标设计
  - 考虑添加辅助任务（如历史位置之间的关系预测）

### 场景2：部分Head是位置偏置，部分是内容相关

```
Head 0, 3-6: 总是关注最后位置（目标导向）
Head 1, 7: 热点位置随内容变化（内容相关）
```

**解读**：
- 这是理想的多头模式
- 位置偏置头：提供任务导向信息
- 内容相关头：捕获历史行为特征
- 建议：
  - 分析内容相关头关注的是哪些行为（购买？收藏？）
  - 验证这些行为是否与预测目标相关

### 场景3：所有Head都是内容相关

```
所有Head的热点位置都随内容变化
```

**解读**：
- 模型完全依赖内容特征
- 可能存在过拟合风险
- 建议：
  - 检查测试集性能
  - 验证位置编码是否正常工作
